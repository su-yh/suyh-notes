# 1. 进入 filebeat 目录

# 2. 清空注册表 data/（关键：强制从头解析，见下方原理）

rm -rf data

# 3. 运行：--once 处理到文件末尾后自动退出；-E 覆盖成紧凑JSON(一行一事件，方便看)
./filebeat -e --once -c /Users/suyunhong/suyh-develop/medcase/filebeat-medcase.yml \
  -E output.console.pretty=false > /tmp/medcase_parse.jsonl 2>/dev/null

# 4. 查看结果
wc -l /tmp/medcase_parse.jsonl     # 事件总数，应为 29355
