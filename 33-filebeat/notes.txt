# 3. 运行：--once 处理到文件末尾后自动退出；-E 覆盖成紧凑JSON(一行一事件，方便看)
./filebeat -e --once -c /Users/suyunhong/suyh-develop/medcase/filebeat-medcase.yml \
  -E output.console.pretty=false > /tmp/medcase_parse.json 2>/dev/null
